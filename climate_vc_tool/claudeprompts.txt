import argparse

def rebuild_index(incremental=False):
    """
    Manual index rebuild process:
    1. Scrape new/updated Crunchbase data (10 min)
    2. Generate embeddings for new startups (15 min on CPU)
    3. Map to simplified verticals (1 min)
    4. Rebuild FAISS index (1 min)
    5. Rebuild BM25 index (30 sec)
    """
    logger.info("Starting manual index rebuild...")
    
    try:
        # Step 1: Scrape Crunchbase
        if incremental:
            logger.info("Incremental update: fetching new/changed startups only")
            new_startups = scrape_crunchbase_incremental()
        else:
            logger.info("Full rebuild: fetching all startups")
            new_startups = scrape_crunchbase_full()
        
        # Step 2: Generate embeddings
        logger.info(f"Generating embeddings for {len(new_startups)} startups")
        embedder = StartupEmbedder()
        
        all_startups = db.execute("SELECT id, short_description, long_description FROM startups").fetchall()
        
        texts = [f"{row[1]} {row[2]}" for row in all_startups]
        embeddings = embedder.generate_embeddings_batch(texts, batch_size=16)
        
        # Save as memory-mapped array
        np.save('data/embeddings/all_embeddings.npy', embeddings)
        
        # Save ID mapping
        id_mapping = [row[0] for row in all_startups]
        with open('data/embeddings/startup_id_mapping.json', 'w') as f:
            json.dump(id_mapping, f)
        
        # Step 3: Map verticals using Crunchbase categories
        logger.info("Mapping verticals from Crunchbase categories")
        for startup in new_startups:
            verticals = map_crunchbase_categories_to_verticals(startup.categories)
            db.execute("""
                UPDATE startups 
                SET primary_vertical = ?, all_verticals = ?, full_text = ?
                WHERE id = ?
            """, (
                verticals[0] if verticals else None,
                json.dumps(verticals),
                f"{startup.name} {startup.short_description} {startup.long_description}",
                startup.id
            ))
        
        # Step 4: Build FAISS index
        logger.info("Building FAISS index")
        faiss_manager = FAISSIndexManager()
        faiss_manager.build_index(embeddings, id_mapping)
        faiss_manager.save_index('data/faiss_index.bin')
        
        # Step 5: Rebuild BM25 index (SQLite FTS5 auto-updates)
        logger.info("Rebuilding BM25 index")
        db.execute("INSERT INTO startups_fts(startups_fts) VALUES('rebuild')")
        
        logger.info("Index rebuild completed successfully!")
        
    except Exception as e:
        logger.error(f"Index rebuild failed: {str(e)}")
        raise

def map_crunchbase_categories_to_verticals(crunchbase_categories):
    """
    Map Crunchbase categories to our simplified verticals
    Uses keyword matching instead of ML classification
    """
    CATEGORY_MAP = {
        'Solar': 'clean_energy',
        'Wind Energy': 'clean_energy',
        'Renewable Energy': 'clean_energy',
        'Energy Storage': 'energy_storage',
        'Battery': 'energy_storage',

        'Green Building': 'built_environment',
        'Recycling': 'circular_economy',
        'Water Treatment': 'water_ocean',
        'Green Finance': 'climate_fintech',
        # ... etc
    }
    
    verticals = []
    for category in crunchbase_categories:
        if category in CATEGORY_MAP:
            vertical_id = CATEGORY_MAP[category]
            if vertical_id not in verticals:
                verticals.append(vertical_id)
    
    return verticals if verticals else ['clean_energy']  # Default fallback

MODEL: sentence-transformers/all-mpnet-base-v2 (CPU-friendly, 768-dim)

from sentence_transformers import SentenceTransformer
import numpy as np

class StartupEmbedder:
    def __init__(self, model_name='sentence-transformers/all-mpnet-base-v2'):
        self.model = SentenceTransformer(model_name, device='cpu')
        
    def generate_embedding(self, text):
        """Generate 768-dim embedding for single text"""
        return self.model.encode(
            text, 
            convert_to_numpy=True, 
            normalize_embeddings=True  # For cosine similarity
        )
    
    def generate_embeddings_batch(self, texts, batch_size=16):
        """
        Batch generation for efficiency
        10K startups takes ~20-30 minutes on CPU
        """
        embeddings = self.model.encode(
            texts,
            batch_size=batch_size,
            show_progress_bar=True,
            convert_to_numpy=True,
            normalize_embeddings=True
        )
        return embeddings.astype('float32')  # Save memory

2.4 FAISS Vector Search

import faiss
import numpy as np
import json

class FAISSIndexManager:
    def __init__(self, dimension=768):
        self.dimension = dimension
        self.index = None
        self.startup_ids = []
        
    def build_index(self, embeddings, startup_ids):
        """
        Build FAISS index from embeddings
        For 10K startups: ~30MB RAM
        """
        # IndexFlatIP for exact cosine similarity (normalized embeddings)
        self.index = faiss.IndexFlatIP(self.dimension)
        self.index.add(embeddings.astype('float32'))
        self.startup_ids = startup_ids
        
        logger.info(f"Built FAISS index with {self.index.ntotal} vectors")
    
    def search(self, query_embedding, top_k=20):
        """Search for similar startups"""
        query = query_embedding.reshape(1, -1).astype('float32')
        similarities, indices = self.index.search(query, top_k)
        
        results = []
        for idx, sim in zip(indices[0], similarities[0]):
            if idx < len(self.startup_ids):
                startup_id = self.startup_ids[idx]
                results.append((startup_id, float(sim)))
        
        return results
    
    def save_index(self, filepath):
        """Save FAISS index and mapping"""
        faiss.write_index(self.index, filepath)
        
        mapping_path = filepath.replace('.bin', '_mapping.json')
        with open(mapping_path, 'w') as f:
            json.dump(self.startup_ids, f)
    
    def load_index(self, filepath):
        """Load FAISS index and ID mapping"""
        self.index = faiss.read_index(filepath)
        
        mapping_path = filepath.replace('.bin', '_mapping.json')
        with open(mapping_path, 'r') as f:
            self.startup_ids = json.load(f)
        
        logger.info(f"Loaded FAISS index with {self.index.ntotal} vectors")

2.5 BM25 Keyword Search

from rank_bm25 import BM25Okapi
import sqlite3

class BM25SearchEngine:
    def __init__(self, db_path):
        self.db = sqlite3.connect(db_path)
        self.startup_texts = []
        self.startup_ids = []
        self.bm25 = None
        
    def build_index(self):
        """Build BM25 index from startup descriptions"""
        cursor = self.db.cursor()
        rows = cursor.execute("""
            SELECT id, name, short_description, long_description 
            FROM startups
        """).fetchall()
        
        self.startup_ids = []
        tokenized_corpus = []
        
        for row in rows:
            startup_id, name, short_desc, long_desc = row
            full_text = f"{name} {short_desc} {long_desc}"
            
            # Simple tokenization (lowercase + split)
            tokens = full_text.lower().split()
            
            self.startup_ids.append(startup_id)
            tokenized_corpus.append(tokens)
        
        self.bm25 = BM25Okapi(tokenized_corpus)
        logger.info(f"Built BM25 index for {len(self.startup_ids)} startups")
    
    def search(self, query, top_k=20):
        """Search using BM25 keyword matching"""
        tokenized_query = query.lower().split()
        scores = self.bm25.get_scores(tokenized_query)
        
        # Get top K results
        top_indices = np.argsort(scores)[::-1][:top_k]
        
        results = []
        for idx in top_indices:
            startup_id = self.startup_ids[idx]
            score = float(scores[idx])
            if score > 0:  # Only return non-zero scores
                results.append((startup_id, score))
        
        return results

2.6 Query Expansion & Understanding

class QueryProcessor:
    def __init__(self):
        # Climate-specific synonym expansion
        self.expansions = {
            'solar': 'solar photovoltaic PV renewable energy sun',
            'battery': 'battery energy storage lithium-ion accumulator',
            'carbon': 'carbon CO2 emissions capture removal sequestration offset',
            'ev': 'electric vehicle EV battery car automotive',
            'wind': 'wind turbine renewable energy offshore onshore',
            'hydrogen': 'hydrogen H2 green hydrogen fuel cell',
            'agriculture': 'agriculture farming agtech food crops',
            'recycling': 'recycling circular waste upcycling reuse',
            'water': 'water desalination purification treatment wastewater',
        }
        
    def expand_query(self, query):
        """Expand query with domain-specific synonyms"""
        query_lower = query.lower()
        expanded_terms = [query]
        
        for keyword, expansion in self.expansions.items():
            if keyword in query_lower:
                expanded_terms.append(expansion)
        
        return ' '.join(expanded_terms)
    
    def extract_filters(self, query):
        """Extract structured filters from natural language query"""
        filters = {}
        
        # Founded year patterns
        import re
        year_match = re.search(r'founded\s+(after|since|before)\s+(\d{4})', query, re.IGNORECASE)
        if year_match:
            operator, year = year_match.groups()
            if operator.lower() in ['after', 'since']:
                filters['founded_year_min'] = int(year)
            elif operator.lower() == 'before':
                filters['founded_year_max'] = int(year)
        
        # Funding patterns
        funding_match = re.search(r'raised\s+over\s+\$?(\d+)([MmBb])', query, re.IGNORECASE)
        if funding_match:
            amount, unit = funding_match.groups()
            multiplier = 1_000_000 if unit.lower() == 'm' else 1_000_000_000
            filters['min_funding_usd'] = int(amount) * multiplier
        
        return filters

2.7 Hybrid Search with Reciprocal Rank Fusion

class HybridSearchEngine:
    def __init__(self, faiss_manager, bm25_engine, embedder):
        self.faiss_manager = faiss_manager
        self.bm25_engine = bm25_engine
        self.embedder = embedder
        self.query_processor = QueryProcessor()
        
    def search(self, query, top_k=10, semantic_weight=0.6):
        """
        Hybrid search combining semantic (FAISS) and keyword (BM25)
        Uses Reciprocal Rank Fusion (RRF) for merging
        """
        # Expand query
        expanded_query = self.query_processor.expand_query(query)
        
        # 1. Semantic search (FAISS)
        query_embedding = self.embedder.generate_embedding(expanded_query)
        semantic_results = self.faiss_manager.search(query_embedding, top_k=20)
        
        # 2. Keyword search (BM25)
        keyword_results = self.bm25_engine.search(query, top_k=20)
        
        # 3. Reciprocal Rank Fusion
        merged_results = self.reciprocal_rank_fusion(
            semantic_results,
            keyword_results,
            semantic_weight=semantic_weight
        )
        
        return merged_results[:top_k]
    
    def reciprocal_rank_fusion(self, semantic_results, keyword_results, k=60, semantic_weight=0.6):
        """
        Merge two ranked lists using RRF
        
        RRF score = semantic_weight * (1 / (k + rank_semantic)) + 
                    (1 - semantic_weight) * (1 / (k + rank_keyword))
        """
        scores = {}
        
        # Add semantic scores
        for rank, (startup_id, sim) in enumerate(semantic_results, start=1):
            scores[startup_id] = semantic_weight * (1 / (k + rank))
        
        # Add keyword scores
        keyword_weight = 1 - semantic_weight
        for rank, (startup_id, score) in enumerate(keyword_results, start=1):
            if startup_id in scores:
                scores[startup_id] += keyword_weight * (1 / (k + rank))
            else:
                scores[startup_id] = keyword_weight * (1 / (k + rank))
        
        # Sort by combined score
        ranked = sorted(scores.items(), key=lambda x: x[1], reverse=True)
        
        return [(startup_id, score) for startup_id, score in ranked]

2.8 Result Diversification

class ResultDiversifier:
    def diversify(self, results, startup_data_map, max_per_vertical=3):
        """
        Ensure diversity across verticals
        Prevents one category from dominating results
        """
        diversified = []
        vertical_counts = {}
        
        for startup_id, score in results:
            startup = startup_data_map[startup_id]
            vertical = startup.get('primary_vertical', 'unknown')
            
            # Check if vertical quota not exceeded
            if vertical_counts.get(vertical, 0) < max_per_vertical:
                diversified.append((startup_id, score))
                vertical_counts[vertical] = vertical_counts.get(vertical, 0) + 1
            
            # Stop when we have enough diverse results
            if len(diversified) >= len(results):
                break
        
        return diversified

2.9 Evaluation Framework

class SearchEvaluator:
    def __init__(self, test_queries_path='config/test_queries.json'):
        """
        test_queries.json format:
        [
            {
                "query": "solar panel companies",
                "relevant_startup_ids": [123, 456, 789],
                "vertical": "clean_energy"
            },
            ...
        ]
        """
        with open(test_queries_path, 'r') as f:
            self.test_queries = json.load(f)
    
    def calculate_ndcg(self, ranked_results, relevant_ids, k=10):
        """
        Calculate Normalized Discounted Cumulative Gain @ K
        """
        dcg = 0.0
        for i, (startup_id, score) in enumerate(ranked_results[:k], start=1):
            if startup_id in relevant_ids:
                # Relevance = 1 if relevant, 0 otherwise
                dcg += 1.0 / np.log2(i + 1)
        
        # Ideal DCG (all relevant items at top)
        idcg = sum(1.0 / np.log2(i + 2) for i in range(min(len(relevant_ids), k)))
        
        return dcg / idcg if idcg > 0 else 0.0
    
    def evaluate(self, search_engine):
        """
        Run evaluation on test queries
        
        Returns:
            dict: {
                'mean_ndcg@10': float,
                'mean_latency_ms': float,
                'p95_latency_ms': float
            }
        """
        ndcg_scores = []
        latencies = []
        
        for test_case in self.test_queries:
            query = test_case['query']
            relevant_ids = test_case['relevant_startup_ids']
            
            # Measure latency
            start = time.time()
            results = search_engine.search(query, top_k=10)
            latency_ms = (time.time() - start) * 1000
            latencies.append(latency_ms)
            
            # Calculate NDCG
            ndcg = self.calculate_ndcg(results, relevant_ids, k=10)
            ndcg_scores.append(ndcg)
        
        return {
            'mean_ndcg@10': np.mean(ndcg_scores),
            'mean_latency_ms': np.mean(latencies),
            'p95_latency_ms': np.percentile(latencies, 95),
            'total_queries': len(self.test_queries)
        }

@app.post("/api/search")
@limiter.limit("30/minute")
async def search_startups(request: Request, search_req: SearchRequest):
    """
    Hybrid semantic + keyword search with result diversification
    """
    import time
    start = time.time()
    
    try:
        # Create cache key
        cache_key = hashlib.md5(
            f"{search_req.query}_{search_req.top_k}".encode()
        ).hexdigest()
        
        # Search (uses LRU cache)
        results = cached_search_internal(cache_key, search_req.query, search_req.top_k)
        
        # Enrich with startup data
        enriched = enrich_search_results(results, app.state.db)
        
        # Apply diversity
        if search_req.enable_diversity:
            startup_map = {r['startup']['id']: r['startup'] for r in enriched}
            results_tuples = [(r['startup']['id'], r['score']) for r in enriched]
            
            diversified_tuples = app.state.diversifier.diversify(
                results_tuples, 
                startup_map,
                max_per_vertical=3
            )
            
            # Reconstruct enriched results in new order
            id_to_enriched = {r['startup']['id']: r for r in enriched}
            enriched = [id_to_enriched[sid] for sid, score in diversified_tuples]
        
        # Apply filters
        if search_req.vertical_filter:
            enriched = [r for r in enriched 
                       if r['startup'].get('primary_vertical') == search_req.vertical_filter]
        
        if search_req.founded_year_min:
            enriched = [r for r in enriched 
                       if r['startup'].get('founded_year', 0) >= search_req.founded_year_min]
        
        if search_req.min_funding_usd:
            enriched = [r for r in enriched 
                       if r['startup'].get('total_funding_usd', 0) >= search_req.min_funding_usd]
        
        processing_time = (time.time() - start) * 1000
        
        return {
            "query": search_req.query,
            "total_results": len(enriched),
            "results": enriched[:search_req.top_k],
            "processing_time_ms": round(processing_time, 2)
        }
        
    except Exception as e:
        logger.error(f"Search failed: {e}")
        raise HTTPException(status_code=500, detail="Search temporarily unavailable")

```python
        4. Merge with RRF
```

```typescript
import { useState, useCallback } from 'react';
import { searchStartups } from '../services/api';
import { SearchRequest, SearchResponse } from '../types';

export const useSearch = () => {
  const [results, setResults] = useState<SearchResponse | null>(null);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const search = useCallback(async (request: SearchRequest) => {
    setLoading(true);
    setError(null);

    try {
      const response = await searchStartups(request);
      setResults(response);
    } catch (err: any) {
      setError(err.response?.data?.detail || 'Search failed. Please try again.');
      setResults(null);
    } finally {
      setLoading(false);
    }
  }, []);

  const clearResults = useCallback(() => {
    setResults(null);
    setError(null);
  }, []);

  return { results, loading, error, search, clearResults };
};
```
